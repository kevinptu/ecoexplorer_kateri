function saveGeometry() {
  var layer_name = geometryName.getValue();
  // var tablePath = folderPath + "/" + layer_name;
  var layers = drawingTools.layers();
  // Get the Earth Engine object (geometry) from the current layer
  var geometry = layers.get(0).getEeObject();
  if(geometry.type().getInfo() === "Point" || geometry.type().getInfo() === "MultiPoint"){
    geometry = geometry.buffer(30) // Creates buffer of 30 m.
  }
  // Calculate the area with an error margin (in square meters)
  var area = geometry.area({maxError: 1}); // Specify error margin in meters
  
// Create a feature with the geometry and the computed area as a property
  var feature = ee.Feature(geometry).set({
    'name': layer_name,
    'area_m2': area
  });

  var tableDownloadUrl = outTable.getDownloadURL('KML', [], layer_name);
  vectorUrlLabel.setUrl(tableDownloadUrl);
  vectorDownloadPanel.style().set({shown: true});

function loadPolygonAssets(pageToken) {
  var params = {pageSize: 1000};
  if (pageToken) {params.pageToken = pageToken;}
  function showAssetError(error) {
    polygonSelector.setPlaceholder('Polygon list unavailable');
    assetStatusLabel.setValue('Could not load saved polygons. You can still draw an area. ' +
      'Check that the published app can access the asset folder and polygon assets. ' +
      String(error && error.message ? error.message : error));
  }
  try {
    ee.data.listAssets(folderPath, params, function(response, error) {
      if (error || !response) {showAssetError(error || 'No response'); return;}
      (response.assets || []).forEach(function(asset) {
        if (asset.type === 'TABLE') {
          assetList.push({name: asset.id.split('/').pop(), assetId: asset.id});
        }
      });
      polygonSelector.items().reset(assetList.map(function(item) {return item.name;}));
      polygonSelector.setDisabled(assetList.length === 0);
      if (response.nextPageToken) {loadPolygonAssets(response.nextPageToken); return;}
      polygonSelector.setPlaceholder(assetList.length ? 'Polygon' : 'No saved polygons');
    });
  } catch (error) {
    showAssetError(error);
  }
}

  selectPolygonPanel.add(ui.Label({value: 'Select polygon', style: {fontSize: '.9vw', fontWeight: 'normal'}}));
  selectPolygonPanel.add(polygonSelector);

// Function to load climate, vegetation, and soil info based on a shapefile
var loadClimateVegetationSoilInfo = function(asset_id, yearofinterest, treeCoverThreshold, displayMaps, shapefileExport) 
{
 
  var bigger_scale = 4638.3;
  var smaller_scale = 30;
  
  // Script to pull climate, vegetation and soil info for an defined by a shapefile
  //
  // ===== Directions ===================================================
  // ===== How to run the script ========================================
  // Step 1. Import a shapefile from your Assets (make sure it's named 'table') -- SEE BELOW
  // Step 2. Enter the desired year of interest above
  // Step 3. Turn the map displays 'on' or 'off'
  // Step 4. Run the script by clicking on the Run button above this panel
  // Step 5. View results in the righthand Console panel
  // Step 6. Optional: To download a shapefile with excluded tree areas go to Tasks > RUN > expand the submitted task > Google Drive and download the shapefile from there
  //
  // ===== How to use a shapefile =======================================
  // Step 1. Delete any existing shapefiles labeled 'var table' at top of this script (hover over and click on garbage can icon)
  // Step 2. Hover over the shapefile name in the Assets list and click on the -> icon to import
  // *Note* - You can import multiple shapefiles but only the one named 'table' will be used
  //
  // ===== How to upload a shapefile to Google Earth Engine =============
  // Step 1. Make sure there are no spaces in the shapefile name
  // Step 2. Make sure that its a .zip file and contains shp, shx, dbf, prj files
  // Step 3. Click on Assets > NEW > Shape files in the lefthand panel
  // Step 4. Navigate to your .zip shapefile by clicking SELECT
  // Step 5. Click on UPLOAD (wait a minute or two for the shapefile to upload...)
  //
  // ===== Data sources used in this script ==================================================
  // TerraClimate for MAT and MAP (https://developers.google.com/earth-engine/datasets/catalog/IDAHO_EPSCOR_TERRACLIMATE)
  // Rangeland Analysis Platform for NPP and grass and shrub cover (https://gee-community-catalog.org/projects/rap/)
  // RAP vegetation-cover-v3 TRE band for tree cover and tree exclusions.
  // Soilgrids for SOC (https://gee-community-catalog.org/projects/isric/)
  // =========================================================================================
  // ===== DON'T CHANGE ANYTHING BELOW =======================================================
  // =========================================================================================

  // ===== Format Start and End Dates For Pulling Data ==========================
  var year = yearofinterest;
  var start = ee.Number(yearofinterest).format("%s").cat('-01-01');
  var startfourbefore = yearofinterest.subtract(4).format("%s").cat('-01-01'); // grab last 5 years of data
  var end = yearofinterest.add(1).format("%s").cat('-01-01'); // Exclusive end: five calendar years.
  var startYear = yearofinterest.subtract(4); // The year of interest
  var startDate = ee.Number(1986).format("%s").cat('-01-01');
  var endYear = yearofinterest; // Five years inclusive of the starting year
  
  // ====== Grab Point Coordinates ===========================================================
  //var table2 = ee.FeatureCollection([ee.Feature(table)]);
  var table;
  if(drawingTools.layers().length() === 1)
    {var layers = drawingTools.layers(); // Access the layers of the drawing tools
    var geometry = layers.get(0).getEeObject(); // Get the Earth Engine object (geometry) from the current layer
    if(geometry.type().getInfo() === "Point" || geometry.type().getInfo() === "MultiPoint")
      {geometry = geometry.buffer(30) // Creates buffer of 30 m.
      bigger_scale = 30}
    var area = geometry.area({maxError: 1}); // Calculate the area with an error margin (in square meters)
    var feature = ee.Feature(geometry).set({'name': 'Layer_0', 'area_m2': area}); // Create a feature with the geometry and the computed area as a property
    table = ee.FeatureCollection(feature);}
  else
    {clearGeometry(); table = ee.FeatureCollection(asset_id);}
  var table2;
  if (table && table.size().getInfo() > 0) {table2 = table;}
  else 
    {var coordinates = geometry.coordinates();
    var point = ee.Geometry.Point(coordinates);
    table2 = ee.FeatureCollection([ee.Feature(point)]);}
  var areaInSqMeters = table2.geometry().area(); // Calculate area in square meters
  var areaInAcres = areaInSqMeters.divide(4046.86); // Convert to acres (1 acre = 4046.86 square meters)

  // ===== Pull Data ===================================================================
  // TerraClimate
  var TerraClimate = ee.ImageCollection('IDAHO_EPSCOR/TERRACLIMATE');
    var TerraClimate5 = TerraClimate.filterDate(startfourbefore, end);
//      .filterBounds(table2).filter(ee.Filter.date(startfourbefore, end));

  // RAP NPP
  var RAP = ee.ImageCollection("projects/rap-data-365417/assets/npp-partitioned-v3");
    var RAPlast5 = RAP.filterDate(startfourbefore, end); 
//      .filterBounds(table2).filter(ee.Filter.date(startfourbefore, end));
      
  var RAPcover = ee.ImageCollection("projects/rap-data-365417/assets/vegetation-cover-v3");
    var RAPcoverLast5 = RAPcover.filterDate(startfourbefore, end);
//      .filterBounds(table2).filter(ee.Filter.date(startfourbefore, end));
  
  // ALOS DSM
  var dataset = ee.ImageCollection('JAXA/ALOS/AW3D30/V3_2')
      .filterBounds(table2);
  var elevation = dataset.select('DSM');

  // SoilGrids
  var bdod_5_15 = ee.Image("projects/soilgrids-isric/bdod_mean").select('bdod_5-15cm_mean').clip(table2); // Bulk Density, 5-15cm (cg/cm^3)
  var isric_ocs = ee.Image("projects/soilgrids-isric/ocs_mean").select('ocs_0-30cm_mean').clip(table2); // SOC stock, 0-30cm (tC/ha)
  var cfvo0_5 = ee.Image("projects/soilgrids-isric/cfvo_mean").select('cfvo_0-5cm_mean').clip(table2); // Volumetric Course Fragment, 0-5cm (cm^3/dm^3 by vol%)
  var cfvo5_15 = ee.Image("projects/soilgrids-isric/cfvo_mean").select('cfvo_5-15cm_mean').clip(table2); // Volumetric Course Fragment, 5-15cm (cm^3/dm^3 by vol%)
  var cfvo15_30 = ee.Image("projects/soilgrids-isric/cfvo_mean").select('cfvo_15-30cm_mean').clip(table2); // Volumetric Course Fragment, 15-30cm (cm^3/dm^3 by vol%)
  var sand0_5 = ee.Image("projects/soilgrids-isric/sand_mean").select('sand_0-5cm_mean').clip(table2); // Sand, 0-5cm (g/kg)
  var sand5_15 = ee.Image("projects/soilgrids-isric/sand_mean").select('sand_5-15cm_mean').clip(table2); // Sand, 5-15cm (g/kg)
  var sand15_30 = ee.Image("projects/soilgrids-isric/sand_mean").select('sand_15-30cm_mean').clip(table2); // Sand, 15-30cm (g/kg)

  // ===== Calculate Spatial Maps of Temporal Averages ==============================================
  var maximumTemperature = TerraClimate5.select('tmmx').reduce(ee.Reducer.mean()).multiply(0.1).clip(table2); // deg C
  var minimumTemperature = TerraClimate5.select('tmmn').reduce(ee.Reducer.mean()).multiply(0.1).clip(table2); // deg C
  var MAP = TerraClimate5.select('pr').reduce(ee.Reducer.sum()).divide(5).clip(table2); // mm/yr
  var MAT = maximumTemperature.add(minimumTemperature).divide(2); // deg C
  var PET = TerraClimate5.select('pet').reduce(ee.Reducer.sum()).divide(5).multiply(0.1).clip(table2); // mm/yr
  var AET = TerraClimate5.select('aet').reduce(ee.Reducer.sum()).divide(5).multiply(0.1).clip(table2); // mm/yr
  var AI = MAP.divide(PET).rename('AI_mean'); // mm/mm
  var afgCover = RAPcoverLast5.select('AFG').reduce(ee.Reducer.mean()).clip(table2); // annual grass and forbs cover (%)
  var pfgCover = RAPcoverLast5.select('PFG').reduce(ee.Reducer.mean()).clip(table2); // perennial grass cover (%)
  var shrCover = RAPcoverLast5.select('SHR').reduce(ee.Reducer.mean()).clip(table2); // shrub cover (%)
  var treCover = RAPcoverLast5.select('TRE').reduce(ee.Reducer.mean()).clip(table2); // tree cover (%)
  var ltrCover = RAPcoverLast5.select('LTR').reduce(ee.Reducer.mean()).clip(table2); // litter cover (%)
  var bgrCover = RAPcoverLast5.select('BGR').reduce(ee.Reducer.mean()).clip(table2); // bare ground (%)
  var afgNPP = RAPlast5.select('afgNPP').reduce(ee.Reducer.mean()).multiply(0.0001).multiply(1000).multiply(10000).divide(2.47105).multiply(44/12).divide(1000000).clip(table2); // tCO2/acre/yr
  var pfgNPP = RAPlast5.select('pfgNPP').reduce(ee.Reducer.mean()).multiply(0.0001).multiply(1000).multiply(10000).divide(2.47105).multiply(44/12).divide(1000000).clip(table2); // tCO2/acre/yr
  var shrNPP = RAPlast5.select('shrNPP').reduce(ee.Reducer.mean()).multiply(0.0001).multiply(1000).multiply(10000).divide(2.47105).multiply(44/12).divide(1000000).clip(table2); // tCO2/acre/yr
  var treNPP = RAPlast5.select('treNPP').reduce(ee.Reducer.mean()).multiply(0.0001).multiply(1000).multiply(10000).divide(2.47105).multiply(44/12).divide(1000000).clip(table2); // tCO2/acre/yr
//  var actualNPP = afgNPP.add(pfgNPP).add(shrNPP).rename('actualNPP_mean'); // tCO2/acre/yr
  var actualNPP = afgNPP.add(pfgNPP).rename('actualNPP_mean'); // tCO2/acre/yr
  var NPPpot = ee.Image.pixelArea();
  var part1 = NPPpot.expression('3000 / (1 + exp(1.315 - 0.119 * MAT))', {'MAT': MAT}); // gDM/m2/yr
  var part2 = NPPpot.expression('3000 * (1 - exp(-0.000664 * MAP))', {'MAP': MAP}); // gDM/m2/yr
  var potentialNPP = part1.min(part2).multiply(0.48).multiply(10000).multiply(44/12).divide(2.47105*1000000).rename('potential_NPP'); // tCO2/acre/yr
  var potentialNPPimprovement = potentialNPP.subtract(actualNPP).where(potentialNPP.subtract(actualNPP).lt(0), 0).rename('potNPPimprovement');
  var potAddSeq = potentialNPPimprovement.multiply(0.2).rename('potAddSeq'); // potential SOC sequestration that is additional
  var cfvo0_30 = cfvo0_5.multiply(5/30).add(cfvo5_15.multiply(10/30)).add(cfvo15_30.multiply(15/30)).divide(1000).multiply(100).rename('cfvo0_30'); // average of % volumetric coarse content from 0-30cm, cm^3/dm^3*(dm*dm*dm/[10cm*10cm*10cm])*(100%)
  var sand0_30 = sand0_5.multiply(5/30).add(sand5_15.multiply(10/30)).add(sand15_30.multiply(15/30)).divide(1000).multiply(100).rename('sand0_30'); // average of % sand content from 0-30cm, g/kg*(kg/1000g)*(100%)
  // Use the same RAP five-year tree-cover mean for statistics and exclusions.
  var tcc = treCover.rename('tree_cover');
  var proj = elevation.first().select(0).projection();
  var slope = ee.Terrain.slope(elevation.mosaic()
     .setDefaultProjection(proj)).clip(table2);

  // ===== Calculate Spatial Averages of 5-yr Means =================================================
    // Annual Average Precipitation over the ROI
  var map_spatialAverage = MAP.reduceRegion({reducer: ee.Reducer.mean(), geometry: table2, scale: bigger_scale, maxPixels: 1e13});
    // Annual Average Air Temperature over the ROI
  var mat_spatialAverage = MAT.reduceRegion({reducer: ee.Reducer.mean(), geometry: table2, scale: bigger_scale, maxPixels: 1e13});
    // Annual Average Actual Evapotranspiration over the ROI
  var aet_spatialAverage = AET.reduceRegion({reducer: ee.Reducer.mean(), geometry: table2, scale: bigger_scale, maxPixels: 1e13});
    // Annual Average Potential Evapotranspiration over the ROI
  var pet_spatialAverage = PET.reduceRegion({reducer: ee.Reducer.mean(), geometry: table2, scale: bigger_scale, maxPixels: 1e13});
    // Annual Average Aridity Index over the ROI
  var ai_spatialAverage = AI.reduceRegion({reducer: ee.Reducer.mean(), geometry: table2, scale: bigger_scale, maxPixels: 1e13});
    // Annual Average Annual Grass & Forb Cover over the ROI
  var afgCover_spatialAverage = afgCover.reduceRegion({reducer: ee.Reducer.mean(), geometry: table2, scale: smaller_scale, maxPixels: 1e10});
    // Annual Average Perennial Grass Cover over the ROI
  var pfgCover_spatialAverage = pfgCover.reduceRegion({reducer: ee.Reducer.mean(), geometry: table2, scale: smaller_scale, maxPixels: 1e10});
    // Annual Average Shrub Cover over the ROI
  var shrCover_spatialAverage = shrCover.reduceRegion({reducer: ee.Reducer.mean(), geometry: table2, scale: smaller_scale, maxPixels: 1e10});
    // Annual Average Tree Cover over the ROI
  var treCover_spatialAverage = treCover.reduceRegion({reducer: ee.Reducer.mean(), geometry: table2, scale: smaller_scale, maxPixels: 1e10});
    // Annual Average Litter Cover over the ROI
  var ltrCover_spatialAverage = ltrCover.reduceRegion({reducer: ee.Reducer.mean(), geometry: table2, scale: smaller_scale, maxPixels: 1e10});
    // Annual Average Bare Ground over the ROI
  var bgrCover_spatialAverage = bgrCover.reduceRegion({reducer: ee.Reducer.mean(), geometry: table2, scale: smaller_scale, maxPixels: 1e10});
    // Annual Average Actual NPP over the ROI
  var NPP_spatialAverage = actualNPP.reduceRegion({reducer: ee.Reducer.mean(), geometry: table2, scale: bigger_scale, maxPixels: 1e13});
    // Annual Average Potential NPP over the ROI
  var NPPpot_spatialAverage = potentialNPP.reduceRegion({reducer: ee.Reducer.mean(), geometry: table2, scale: bigger_scale, maxPixels: 1e13});
    // Annual Average Potential NPP Improvement over the ROI
  var potentialNPPimprovement_spatialAverage = potentialNPPimprovement.reduceRegion({reducer: ee.Reducer.mean(), geometry: table2, scale: bigger_scale, maxPixels: 1e13});
    // Annual Average Potential Additional SOC Sequestration over the ROI
  var potAddSeq_spatialAverage = potAddSeq.reduceRegion({reducer: ee.Reducer.mean(), geometry: table2, scale: bigger_scale, maxPixels: 1e13});
    // SoilGrids - OCS, 0-30cm over the ROI
  var ocs_spatialAverage = isric_ocs.reduceRegion({reducer: ee.Reducer.mean(), geometry: table2, scale: 250, maxPixels: 1e13});
    // SoilGrids - Volumetric Coarse Fraction, 0-30cm over the ROI
  var cfvo0_30_spatialAverage = cfvo0_30.reduceRegion({reducer: ee.Reducer.mean(), geometry: table2, scale: 250, maxPixels: 1e13});
    // SoilGrids - Sand, 0-30cm over the ROI
  var sand0_30_spatialAverage = sand0_30.reduceRegion({reducer: ee.Reducer.mean(), geometry: table2, scale: 250, maxPixels: 1e13});
    // RAP tree cover spatial average over the ROI.
  var treeCover_spatialAverage = tcc.reduceRegion({reducer: ee.Reducer.mean(), geometry: table2.geometry(), scale: smaller_scale, maxPixels: 1e13});
    // Average Slope over the ROI
  var slope_spatialAverage = slope.reduceRegion({reducer: ee.Reducer.mean(), geometry: table2, scale: smaller_scale, maxPixels: 1e13});
  var percentGrade = ee.Number(slope_spatialAverage.get('slope')).multiply(Math.PI).divide(180).tan().multiply(100);

//var RAP = ee.ImageCollection("projects/rap-data-365417/assets/npp-partitioned-v3");

// Function to calculate the mean monthly air temperature
function MMT_calc(image) 
  {var MMT = image.select(['tmmx', 'tmmn']).reduce(ee.Reducer.mean()); // Average the two bands
  MMT = MMT
    .multiply(0.1)  // MMT scalar (deg C)
    .rename('MMT'); // Rename the result
  return image.addBands(MMT);} // Add the 'MMT' band back to the image
  
// Calculate MAT for each year
function MAT_calc(imageCollection) {
  // Apply MMT_calc to calculate monthly mean temperature
  var withMMT = imageCollection.map(MMT_calc);
  // Group by year
  var years = ee.List.sequence(1986, 2025); // Example years
  var annualMAT = years.map(function(year) {
    var yearCollection = withMMT.filter(ee.Filter.calendarRange(year, year, 'year'));
    var MAT = yearCollection.select('MMT')
      .reduce(ee.Reducer.mean())
      .rename('MAT') // Rename the band explicitly to 'MAT'
      .set('system:time_start', ee.Date.fromYMD(year, 1, 1).millis()); // Set time
    return MAT;
  });
  return ee.ImageCollection(annualMAT);
}

// Calculate MAP for each year
function MAP_calc(imageCollection) {
  // Apply MAP_calc to calculate monthly mean temperature
  var withMAP = TerraClimate.select('pr');
  // Group by year
  var years = ee.List.sequence(1986, 2025); // Example years
  var annualMAP = years.map(function(year) {
    var yearCollection = withMAP.filter(ee.Filter.calendarRange(year, year, 'year'));
    var MAP = yearCollection.select('pr')
      .reduce(ee.Reducer.mean())
      .rename('MAP') // Rename the band explicitly to 'MAP'
      .set('system:time_start', ee.Date.fromYMD(year, 1, 1).millis()); // Set time
    return MAP;
  });
  return ee.ImageCollection(annualMAP);
}
